let rec fold_left_cps (f : 'a -> 'b -> ('a -> 'c) -> 'c) (acc : 'a) (xs : 'b list) (finFun : 'a -> 'c) : 'c =
    match xs with
    | [] -> finFun acc
    | x :: xs -> f acc x (fun new_acc -> fold_left_cps f new_acc xs finFun)
